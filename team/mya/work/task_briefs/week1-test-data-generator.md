```python
"""Generate synthetic test data for backtester testing."""

import pandas as pd
import numpy as np
from datetime import datetime, timedelta

def generate_test_data(
    n_games: int = 100,
    start_date: str = "2026-01-01",
    home_win_rate: float = 0.55,  # Slight home advantage
    seed: int = 42
) -> pd.DataFrame:
    """
    Generate fake game data with known outcomes.

    Args:
        n_games: Number of games to generate
        start_date: Start date for games
        home_win_rate: Probability home team wins
        seed: Random seed for reproducibility

    Returns:
        DataFrame matching backtester input format
    """
    np.random.seed(seed)

    teams = [
        "Los Angeles Lakers", "Boston Celtics", "Golden State Warriors",
        "Miami Heat", "Phoenix Suns", "Milwaukee Bucks",
        "Philadelphia 76ers", "Denver Nuggets", "Dallas Mavericks",
        "Brooklyn Nets"
    ]

    data = []
    start = datetime.strptime(start_date, "%Y-%m-%d")

    for i in range(n_games):
        # Pick random home/away teams
        home_idx, away_idx = np.random.choice(len(teams), 2, replace=False)
        home_team = teams[home_idx]
        away_team = teams[away_idx]

        # Determine outcome
        home_win = 1 if np.random.random() < home_win_rate else 0

        # Generate odds (slightly favor the eventual winner)
        if home_win:
            home_odds = np.random.uniform(1.7, 2.2)
            away_odds = np.random.uniform(1.8, 2.3)
        else:
            home_odds = np.random.uniform(1.8, 2.3)
            away_odds = np.random.uniform(1.7, 2.2)

        data.append({
            'timestamp': start + timedelta(days=i // 5, hours=19),
            'game': f"{home_team} vs {away_team}",
            'home_team': home_team,
            'away_team': away_team,
            'home_odds': round(home_odds, 3),
            'away_odds': round(away_odds, 3),
            'home_win': home_win,
        })

    return pd.DataFrame(data)

def generate_perfect_test(n_games: int = 20) -> pd.DataFrame:
    """
    Generate test data where outcome is predictable.
    Even games = home wins, odd games = away wins.
    Useful for testing backtester logic.
    """
    data = []
    start = datetime(2026, 1, 1, 19, 0)

    for i in range(n_games):
        home_win = 1 if i % 2 == 0 else 0

        data.append({
            'timestamp': start + timedelta(days=i),
            'game': f"Team{i*2} vs Team{i*2+1}",
            'home_team': f"Team{i*2}",
            'away_team': f"Team{i*2+1}",
            'home_odds': 2.0,
            'away_odds': 2.0,
            'home_win': home_win,
        })

    return pd.DataFrame(data)

def generate_edge_cases() -> dict:
    """Generate edge case test data."""
    return {
        'empty': pd.DataFrame(),
        'single_game': generate_test_data(n_games=1),
        'all_home_wins': generate_test_data(n_games=20, home_win_rate=1.0),
        'all_away_wins': generate_test_data(n_games=20, home_win_rate=0.0),
        'high_odds': generate_test_data(n_games=10).assign(
            home_odds=lambda x: x['home_odds'] * 5,
            away_odds=lambda x: x['away_odds'] * 5
        ),
    }

if __name__ == "__main__":
    # Generate and save test data
    df = generate_test_data(100)
    df.to_csv("data/test_games.csv", index=False)
    print(f"Generated {len(df)} test games")
    print(f"Home win rate: {df['home_win'].mean():.1%}")
    print(df.head())
```

### Perfect Test Data
Use `generate_perfect_test()` for deterministic testing:
- Even-indexed games: home wins
- Odd-indexed games: away wins
- All odds are 2.0 (even money)

### Edge Cases
Use `generate_edge_cases()` for:
- Empty data
- Single game
- All home wins
- All away wins
- Extreme odds
